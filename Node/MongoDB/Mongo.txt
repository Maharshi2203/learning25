1.Find all employees whose salary is greater than 40000.
ans- db.employees.find({salary:{$gt:40000}})

2.Find employees whose department is Developer and city is Ahmedabad.
ans-db.employees.find({$and:[{department:"Developer"},{city:"Ahemdabad"}]})

3.Find employees whose age is between 24 and 29.
ans- db.employees.find({ age: { $gte: 24, $lte: 29 } })

4.Find employees whose city is either Mumbai or Pune.
ans - db.employees.find($or:[{city:"Mumbai"},{city:"pune"}])

5.Find employees whose department is not HR.
ans- db.employees.find({department:{$ne:HR}})

6.Find employees whose status is true and salary greater than 50000.
ans- db.employees.find({$or:[{status:true},{salary:{ $gt:50000}}]})

7.Sort all employees by salary in descending order.
ans- db.employees.find().sort({salary:1})

8.Find first 3 employees after skipping 2 records.
ans- db.employees.find().skip(2).limit(3)

9.Find employees whose name is in ["Raj","Priya","Riya","Amit"].
ans- db.employees.find({name:{$in:["Raj","Priya","Riya","Amit"]}})

10.Find employees whose department is Developer or Tester and age less than 28.
ans- db.employees.find({$and:[{$or:[{department:"Developer"},{department:"Tester"}]},{age:{$lt:28}}]})
